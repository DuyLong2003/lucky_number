const httpStatus = require('http-status');
const { ${name} } = require('../models');
const ApiError = require('../utils/ApiError');

/**

 * @param {Object} ${lname}Body
 * @returns {Promise<${name}>}
 */
const create${name} = async (${lname}Body) => {
  const ${lname} = await ${name}.create(${lname}Body);

/**

/**

 * @param {ObjectId} id
 * @returns {Promise<${name}>}
 */
const get${name}ById = async (id) => ${name}.findById(id);

/**

 * @param {ObjectId} ${lname}Id
 * @param {Object} updateBody
 * @returns {Promise<${name}>}
 */
const update${name}ById = async (${lname}Id, updateBody) => {

  Object.assign(${lname}, updateBody);

/**
 * Delete ${lname} by id
 * @param {ObjectId} ${lname}Id
 * @returns {Promise<${name}>}
 */
const delete${name}ById = async (${lname}Id) => {
